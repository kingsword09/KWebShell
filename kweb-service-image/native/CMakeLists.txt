cmake_minimum_required(VERSION 3.21)
project(kweb_service_image_native VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT DEFINED KWEB_PROJECT_ARCH OR KWEB_PROJECT_ARCH STREQUAL "")
  message(FATAL_ERROR "KWEB_PROJECT_ARCH must identify the configured desktop architecture.")
endif()

if(APPLE)
  enable_language(OBJCXX)
  set(KWEB_IMAGE_PLATFORM_SOURCE src/image_mac.mm)
elseif(WIN32)
  set(KWEB_IMAGE_PLATFORM_SOURCE src/image_win.cc)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GDK_PIXBUF REQUIRED IMPORTED_TARGET gdk-pixbuf-2.0)
  set(KWEB_IMAGE_PLATFORM_SOURCE src/image_linux.cc)
else()
  message(FATAL_ERROR "KWebNativeImage supports only Windows, macOS, and Linux.")
endif()

if((APPLE AND NOT KWEB_PROJECT_ARCH STREQUAL "arm64") OR
   (NOT APPLE AND NOT KWEB_PROJECT_ARCH STREQUAL "x86_64"))
  message(FATAL_ERROR "The image provider is published only for macOS arm64, Windows x64, and Linux x64.")
endif()

set(KWEB_CONTRACT_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/contract")
add_library(kweb_image_abi SHARED src/image_abi.cc ${KWEB_IMAGE_PLATFORM_SOURCE})
target_include_directories(kweb_image_abi PRIVATE include src)
target_compile_definitions(kweb_image_abi PRIVATE KWEB_IMAGE_ABI_BUILD)
if(MSVC)
  target_compile_options(kweb_image_abi PRIVATE /W4 /WX /utf-8 /EHsc)
  target_link_libraries(kweb_image_abi PRIVATE gdi32)
else()
  target_compile_options(kweb_image_abi PRIVATE -Wall -Wextra -Wpedantic -Werror $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
endif()
if(APPLE)
  target_link_libraries(kweb_image_abi PRIVATE "-framework CoreGraphics" "-framework CoreFoundation")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(kweb_image_abi PRIVATE PkgConfig::GDK_PIXBUF)
endif()
set_target_properties(kweb_image_abi PROPERTIES
  OUTPUT_NAME kwebshell_image
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN YES
  LIBRARY_OUTPUT_DIRECTORY "${KWEB_CONTRACT_OUTPUT_DIR}"
  RUNTIME_OUTPUT_DIRECTORY "${KWEB_CONTRACT_OUTPUT_DIR}"
)

enable_testing()
find_package(Threads REQUIRED)
add_executable(kweb_image_tests tests/image_tests.cc ${KWEB_IMAGE_PLATFORM_SOURCE})
target_include_directories(kweb_image_tests PRIVATE include src)
target_link_libraries(kweb_image_tests PRIVATE kweb_image_abi Threads::Threads)
if(APPLE)
  target_link_libraries(kweb_image_tests PRIVATE "-framework CoreGraphics" "-framework CoreFoundation")
elseif(WIN32)
  target_link_libraries(kweb_image_tests PRIVATE gdi32)
else()
  target_link_libraries(kweb_image_tests PRIVATE PkgConfig::GDK_PIXBUF)
endif()
if(MSVC)
  target_compile_options(kweb_image_tests PRIVATE /W4 /WX /utf-8 /EHsc /UNDEBUG)
else()
  target_compile_options(kweb_image_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
set_target_properties(kweb_image_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${KWEB_CONTRACT_OUTPUT_DIR}")
if(APPLE)
  set_target_properties(kweb_image_tests PROPERTIES BUILD_RPATH "${KWEB_CONTRACT_OUTPUT_DIR}")
elseif(UNIX)
  set_target_properties(kweb_image_tests PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
add_test(NAME kweb_image_tests COMMAND kweb_image_tests)
set_tests_properties(kweb_image_tests PROPERTIES LABELS "abi;native;image" TIMEOUT 30)
